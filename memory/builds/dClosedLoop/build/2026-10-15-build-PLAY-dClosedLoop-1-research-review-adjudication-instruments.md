## `parse.py`

"""I1: parse every '## Appendix — every finding' table under memory/builds/*/reviews/.
Writes rows.json: one dict per finding, with report, kind, round, N, batch (reconstructed:
contiguous chunks of ceil(N/5), tier2-review.js:1284). A row a raw line break cut is re-joined.
Refuses on an unparseable verdict or a short row."""
import glob, json, math, re, sys, os
BS = chr(92); NL = chr(10); CR = chr(13); SOH = chr(1)
ROW = re.compile(r'^[|] [0-9]+ [|]')
root = sys.argv[1].replace(BS, '/'); out = sys.argv[2]
rows = []; reports = []
for p in sorted(glob.glob(root + '/memory/builds/*/reviews/*.md')):
    p = p.replace(BS, '/')
    t = open(p, encoding='utf-8', newline='').read()
    if '## Appendix — every finding' not in t: continue
    tail = t.split('## Appendix — every finding', 1)[1]
    head = t[:3000]
    name = os.path.basename(p)
    kind = 'spec' if ('spec-audit' in name or re.search(r'[*][*]Serves:[*][*] *spec-audit', head)) else 'diff'
    m = re.search(r'round[- ]?([0-9]+)', name); rnd = int(m.group(1)) if m else 1
    joined = []
    for line in tail.split(NL):
        line = line.rstrip(CR)
        if ROW.match(line) or not joined or line.startswith('|') or line.startswith('#') or not line.strip():
            joined.append(line)
        else:
            joined[-1] += ' ' + line
    rr = []
    for line in joined:
        if not ROW.match(line): continue
        s = line.strip().replace(BS + '|', SOH)
        cells = [c.strip().replace(SOH, '|') for c in s.split('|')[1:-1]]
        if len(cells) < 8: sys.exit(f'{p}: short row {line[:80]}')
        fid, lens, ref, sev, skep, verdict, reason, fixv = cells[:8]
        cls = cells[8] if len(cells) > 8 else ''
        if verdict not in ('confirmed', 'refuted', 'uncertain', 'unverified'):
            sys.exit(f'{p}: id {fid} verdict {verdict!r}')
        rr.append(dict(report=p[len(root) + 1:], build=p.split('/')[-3], kind=kind, round=rnd, id=int(fid), lens=lens,
                       ref=ref, sev=sev, skep=skep, verdict=verdict, reason=reason, fix=fixv, classes=cls))
    if not rr: sys.exit(f'{p}: appendix with no rows')
    N = len(rr); size = math.ceil(N / 5)
    for r in rr:
        r['N'] = N; r['batch'] = (r['id'] - 1) // size
    ids = sorted(r['id'] for r in rr)
    if ids != list(range(1, N + 1)): print(f'WARN {p}: ids not 1..N', file=sys.stderr)
    rows += rr; reports.append((p[len(root) + 1:], kind, rnd, N))
json.dump(rows, open(out, 'w', encoding='utf-8'), indent=0)
print(f'reports {len(reports)}  rows {len(rows)}  spec {sum(1 for r in reports if r[1] == "spec")}  diff {sum(1 for r in reports if r[1] == "diff")}')
for r in reports: print(f'{r[1]:4} r{r[2]} N={r[3]:3}  {r[0]}')
