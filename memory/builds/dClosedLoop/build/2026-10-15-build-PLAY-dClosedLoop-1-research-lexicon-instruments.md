root=sys.argv[1]; d=json.load(open(sys.argv[2]))
py=[o for o in d if o["path"].endswith(".py")]

for path in sorted({o["path"] for o in py}):

    def walk(n,stack):
        for c in ast.iter_child_nodes(n):
            if isinstance(c,ast.ClassDef):
                walk(c,stack+[("class",c.name,[ast.unparse(b) for b in c.bases])])

"""Read-only prototype: which Python method definitions are FORCED by an out-of-corpus base.
argv: repo root, offenders.tsv. Imports only stdlib/installed modules named by the corpus' imports;
never imports a corpus module (in-repo bases are resolved by AST only)."""
import ast, importlib, subprocess, sys, json, collections
root, offp = sys.argv[1], sys.argv[2]
files = [f for f in subprocess.run(["git","ls-files","*.py"],cwd=root,capture_output=True,text=True).stdout.split() ]
local_tops = set()
for f in files:
    parts=f.split("/"); local_tops.add(parts[-1][:-3])
mods = {}   # file -> (tree, alias map)
classes = {}  # (file, clsname) -> (method names set, base exprs)
def aliases(tree):
    m={}
    for n in tree.body if False else ast.walk(tree):
        if isinstance(n, ast.Import):
            for a in n.names: m[(a.asname or a.name).split(".")[0]] = (a.asname and a.name) or a.name.split(".")[0]
        elif isinstance(n, ast.ImportFrom) and n.level==0 and n.module:
            for a in n.names: m[a.asname or a.name] = n.module + "." + a.name
    return m
def dotted(e):
    if isinstance(e, ast.Name): return e.id
    if isinstance(e, ast.Attribute):
        d=dotted(e.value); return d and d+"."+e.attr
    if isinstance(e, ast.Subscript): return dotted(e.value)   # Generic[T]
    return None

    except Exception as ex: print("PARSE",f,ex,file=sys.stderr); continue

    for n in ast.walk(t):
        if isinstance(n, ast.ClassDef):
            meths={c.name for c in n.body if isinstance(c,(ast.FunctionDef,ast.AsyncFunctionDef))}
            classes[(f,n.name)]=(meths,[dotted(b) for b in n.bases],n)
by_name=collections.defaultdict(list)
for (f,c) in classes: by_name[c].append(f)
ext_cache={}
def ext_class(path):
    if path in ext_cache: return ext_cache[path]
    mod,_,attr=path.rpartition("."); r=None
    if mod and mod.split(".")[0] not in local_tops:
        try:
            o=importlib.import_module(mod)
            for p in attr.split("."): o=getattr(o,p)
            r=o if isinstance(o,type) else None
        except Exception: r="UNRESOLVED"

    """-> ('ext', cls) | ('local', (file,cls)) | ('unresolved', b)"""
    if b is None: return ("unresolved",None)

        if full.split(".")[0] in local_tops:
            last=full.split(".")[-1]
            for g in by_name.get(last,[]): return ("local",(g,last))
            return ("unresolved",full)

    c=ext_class(b)
    return ("ext",c) if isinstance(c,type) else ("unresolved",b)

    """Who first defines `name` above class (f,cls)? -> ('ext', qualname) | ('local', (f,c)) | None | 'unresolved'"""
    seen=seen or set(); unres=False

        if kind=="ext":
            if hasattr(v,name):
                owner=next((k for k in v.__mro__ if name in vars(k)),v)
                return ("ext", f"{owner.__module__}.{owner.__qualname__}")

            seen.add(v)

            r=defining(v[0],v[1],name,seen)
            if r and r!="unresolved": return r
            unres|= r=="unresolved"
        elif kind=="unresolved": unres=True

    if name in vars(object): return ("ext","builtins.object")

    sel={b:byname.get(b,set()) for b in bases}
    rows=sum(len(v) for v in sel.values()); distinct=set().union(*sel.values())
    print(f"{name}: {len(cl)} classes in package; selected {sum(1 for v in sel.values() if v)}/{len(bases)} bases found; (base,method) rows {rows}; distinct names {len(distinct)}")
