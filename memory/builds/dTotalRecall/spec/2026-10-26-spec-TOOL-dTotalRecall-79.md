## 4. Design

### The derivation

```bash
# run from the repository root; SWEEP holds unit 78's fenced block, LIVE unit 77's two live lines
cat "$SWEEP" "$LIVE" > "$SCRATCH/spans-in.txt"
python - "$SCRATCH/spans-in.txt" <<'SPANS'
import re, sys
order, rc, checks = [], {}, {}
for raw in open(sys.argv[1], encoding='utf-8'):
    line = raw.strip()
    m = re.match(r'^(\S+)(?: [0-9a-f]{40})? rc=(\d+)$', line)
    if m:
        order.append(m.group(1)); rc[m.group(1)] = int(m.group(2)); continue
    m = re.match(r'^(\S+)(?: [0-9a-f]{40})? MANIFEST check (\d+) FAILED', line)
    if m:
        checks.setdefault(m.group(1), set()).add(m.group(2)); continue
    m = re.match(r'^(\S+) red ', line)
    if m:
        order.append(m.group(1)); rc[m.group(1)] = None
spans, cur = [], None
for u in order:
    if rc[u] is None:
        print('unresolved', u)
    elif rc[u]:
        cur = cur or [u, u, set()]
        cur[1] = u
        cur[2] |= checks.get(u, set())
    elif cur:
        spans.append(cur + [u]); cur = None
if cur:
    spans.append(cur + [None])
for first, last, cks, closer in spans:
    print('span', first, last, 'checks', ','.join(sorted(cks)) or 'none-printed', 'closed-by', closer or 'OPEN')
print('spans', len(spans))
SPANS
```

Run once with `$LIVE` and once with an empty `$LIVE`. The first run's last span line must be the
staged red, `span live live checks 9 closed-by OPEN`. The second run's spans are the leg's history.
