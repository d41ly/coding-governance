```python
def derive_proof_path(top, commit, ending):
    """The one record `commit` holds under `PROOF_DIR` whose name ends `ending`, repo-relative; refused
    unless exactly one does. Read from the commit, so nothing in today's tree moves the answer."""
    proc = subprocess.run(["git", "-C", str(top), "ls-tree", "--name-only", commit, PROOF_DIR + "/"],
                          capture_output=True, text=True, encoding="utf-8", errors="replace", check=False)
    if proc.returncode:
        raise Refusal(f"proof: `git ls-tree {commit}` exited {proc.returncode}: {proc.stderr.strip() or 'no stderr'}")
    found = [p for p in proc.stdout.splitlines() if p.endswith(ending)]
    if len(found) != 1:
        raise Refusal(f"proof: {len(found)} record(s) under {PROOF_DIR} at {commit} end {ending}, not one")
    return found[0]
```

```python
def scan_arm_labels(root, specs, files):
    """THE ARM-LABELS JOIN (TOOL-dFairHearing-48 S9). -> (hits, carriers, graded)."""
    hits, carriers, graded = [], 0, 0
    for f in specs:
        text = render_unfenced((root / f).read_bytes().decode("utf-8", "replace").replace("\r\n", "\n"))
        bullets = ARM_LABEL_BULLET.findall(text)
        if not bullets:
            continue
        carriers += 1
        arms = read_arm_lines(text)
        for label, ac in ARM_LABEL.findall(" ".join(bullets)):
            graded += 1
            suites = [p for p, fields in arms if any(ac in toks for toks in fields) and p in files]
            said = re.compile(re.escape(label) + r"(?![0-9])")
            if not any(said.search((root / p).read_bytes().decode("utf-8", "replace")) for p in suites):

    return hits, carriers, graded
```
